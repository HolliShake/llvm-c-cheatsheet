LLVM-C API · reference & build guide

Everything the C API gives you —
and everything it doesn't.

A working reference for llvm-c/*.h: IR construction, verification, optimization, codegen, and JIT — plus the full parts list for a compiler that actually ships.

$ clang myc.c $(llvm-config --cflags --libs core native) -o myc

01 Setup

Locate flags with llvm-config, then pull in the headers you need.

# Find flags/libs for your LLVM install
llvm-config --version
llvm-config --cflags
llvm-config --libs core analysis executionengine mcjit interpreter native
llvm-config --system-libs

# Compile example
clang myc.c $(llvm-config --cflags --libs core native) $(llvm-config --system-libs) -o myc

Core headers

#include <llvm-c/Core.h>          // IR building: types, values, instructions
#include <llvm-c/Analysis.h>      // module/function verification
#include <llvm-c/Target.h>        // target init
#include <llvm-c/TargetMachine.h> // codegen, object emission
#include <llvm-c/Transforms/PassBuilder.h> // new-PM optimization passes
#include <llvm-c/ExecutionEngine.h> // JIT execution (or llvm-c/Orc.h / OrcEE.h / LLJIT.h)
#include <llvm-c/BitWriter.h>     // emit .bc bitcode
#include <llvm-c/DebugInfo.h>     // DWARF debug info (DIBuilder-C)
#include <llvm-c/Error.h>         // LLVMErrorRef handling

02 The Big Four Objects

Everything else in this document is built on top of these four handles.

ObjectTypePurpose
ContextLLVMContextRefOwns types/constants; isolates compilations (thread-safety unit)
ModuleLLVMModuleRefA translation unit: globals, functions, IR
BuilderLLVMBuilderRefCursor that inserts instructions into basic blocks
Basic BlockLLVMBasicBlockRefStraight-line instruction sequence ending in a terminator
LLVMContextRef ctx = LLVMContextCreate();
LLVMModuleRef  mod = LLVMModuleCreateWithNameInContext("mymodule", ctx);
LLVMBuilderRef bld = LLVMCreateBuilderInContext(ctx);

// ... build IR ...

LLVMDisposeBuilder(bld);
LLVMDisposeModule(mod);
LLVMContextDispose(ctx);

04 Functions & Basic Blocks

LLVMValueRef fn = LLVMAddFunction(mod, "add", fnTy);
LLVMSetLinkage(fn, LLVMExternalLinkage);

LLVMBasicBlockRef entry = LLVMAppendBasicBlockInContext(ctx, fn, "entry");
LLVMPositionBuilderAtEnd(bld, entry);

LLVMValueRef a = LLVMGetParam(fn, 0);
LLVMValueRef b = LLVMGetParam(fn, 1);
LLVMSetValueName2(a, "a", 1);

LLVMValueRef sum = LLVMBuildAdd(bld, a, b, "sum");
LLVMBuildRet(bld, sum);
Other useful block ops: LLVMInsertBasicBlock, LLVMMoveBasicBlockAfter, LLVMDeleteBasicBlock, LLVMGetBasicBlockTerminator.

06 Extended Operators

Arithmetic & Bitwise

// Arithmetic
LLVMValueRef mul = LLVMBuildMul(bld, a, b, "mul");
LLVMValueRef mod = LLVMBuildSRem(bld, a, b, "mod"); // SRem (signed), URem (unsigned), FRem (float)
LLVMValueRef add = LLVMBuildAdd(bld, a, b, "add");
LLVMValueRef sub = LLVMBuildSub(bld, a, b, "sub");

// Bitwise
LLVMValueRef shl  = LLVMBuildShl(bld, a, b, "shl");
LLVMValueRef ashr = LLVMBuildAShr(bld, a, b, "ashr"); // Arithmetic shift right (preserves sign)
LLVMValueRef lshr = LLVMBuildLShr(bld, a, b, "lshr"); // Logical shift right (fills 0s)
LLVMValueRef band = LLVMBuildAnd(bld, a, b, "and");
LLVMValueRef bor  = LLVMBuildOr(bld, a, b, "or");
LLVMValueRef bxor = LLVMBuildXor(bld, a, b, "xor");

// Unary
LLVMValueRef neg = LLVMBuildNeg(bld, a, "neg"); // Integer negation (-a)
LLVMValueRef not = LLVMBuildNot(bld, a, "not"); // Bitwise NOT (~a)

Comparisons (Integer)

// Use LLVMBuildFCmp and LLVMRealPredicate for floating point
LLVMValueRef cmp_lt = LLVMBuildICmp(bld, LLVMIntSLT, a, b, "cmp_lt"); // <  (Signed Less Than)
LLVMValueRef cmp_le = LLVMBuildICmp(bld, LLVMIntSLE, a, b, "cmp_le"); // <= (Signed Less Equal)
LLVMValueRef cmp_gt = LLVMBuildICmp(bld, LLVMIntSGT, a, b, "cmp_gt"); // >  (Signed Greater Than)
LLVMValueRef cmp_ge = LLVMBuildICmp(bld, LLVMIntSGE, a, b, "cmp_ge"); // >= (Signed Greater Equal)
LLVMValueRef cmp_eq = LLVMBuildICmp(bld, LLVMIntEQ,  a, b, "cmp_eq"); // ==
LLVMValueRef cmp_ne = LLVMBuildICmp(bld, LLVMIntNE,  a, b, "cmp_ne"); // !=

Logical AND / OR (Short-circuiting)

Logical operators like && and || in C require short-circuiting. You cannot just use LLVMBuildAnd. You must generate basic blocks to skip the right-hand evaluation if the left-hand determines the result.

// Example: a && b
LLVMBasicBlockRef lhs_true = LLVMAppendBasicBlockInContext(ctx, fn, "and.rhs");
LLVMBasicBlockRef merge = LLVMAppendBasicBlockInContext(ctx, fn, "and.merge");

// Assume 'a' is an i1 condition
LLVMBuildCondBr(bld, a, lhs_true, merge);

// Evaluate RHS
LLVMPositionBuilderAtEnd(bld, lhs_true);
// ... generate 'b' ...
LLVMBasicBlockRef rhs_end = LLVMGetInsertBlock(bld); // block might have changed
LLVMBuildBr(bld, merge);

// Merge
LLVMPositionBuilderAtEnd(bld, merge);
LLVMValueRef phi = LLVMBuildPhi(bld, i1, "and.res");
LLVMValueRef false_val = LLVMConstInt(i1, 0, 0);
LLVMAddIncoming(phi, &false_val, &original_block, 1);
LLVMAddIncoming(phi, &b, &rhs_end, 1);

07 Access, Calls & Assignments

Member Access & Indexing

// Array Indexing (e.g., arr[i])
LLVMValueRef indices[] = { LLVMConstInt(i32, 0, 0), index_val };
LLVMValueRef gep = LLVMBuildGEP2(bld, array_ty, array_ptr, indices, 2, "arr_idx");
LLVMValueRef loaded = LLVMBuildLoad2(bld, elem_ty, gep, "val");

// Struct Member Access (e.g., struct.field)
// The index MUST be a constant for StructGEP
LLVMValueRef field_ptr = LLVMBuildStructGEP2(bld, struct_ty, struct_ptr, field_index, "field");

Function Calls

// Direct call
LLVMValueRef args[] = { arg1, arg2 };
LLVMValueRef call = LLVMBuildCall2(bld, fn_ty, fn, args, 2, "call_res");

// Postfix operators like x++ / x-- require a Load, Add/Sub 1, Store, returning old val.

Assignments & Compound Assignments

// Normal Assignment: a = b
LLVMBuildStore(bld, b_val, a_ptr);

// Compound Assignment: a += b (Load -> Add -> Store)
LLVMValueRef curr = LLVMBuildLoad2(bld, i32, a_ptr, "load");
LLVMValueRef res = LLVMBuildAdd(bld, curr, b_val, "add");
LLVMBuildStore(bld, res, a_ptr); // %=, &=, etc. follow the exact same load-op-store pattern.

08 Declarations

// Struct Declaration (Named, allows recursive definitions)
LLVMTypeRef my_struct = LLVMStructCreateNamed(ctx, "MyStruct");
LLVMTypeRef fields[] = { LLVMInt32Type(ctx), LLVMPointerType(ctx, 0) };
LLVMStructSetBody(my_struct, fields, 2, /*isPacked=*/0);

// Function Declaration (Prototype without body)
LLVMTypeRef param_types[] = { i32, i32 };
LLVMTypeRef fn_ty = LLVMFunctionType(i32, param_types, 2, /*isVarArg=*/0);
LLVMValueRef my_fn = LLVMAddFunction(mod, "my_func", fn_ty);
// Note: To make it an definition, append a BasicBlock. Otherwise, it remains a declaration.

09 Control Flow

If / Else Statement

LLVMBasicBlockRef then_bb = LLVMAppendBasicBlockInContext(ctx, fn, "then");
LLVMBasicBlockRef else_bb = LLVMAppendBasicBlockInContext(ctx, fn, "else");
LLVMBasicBlockRef merge_bb = LLVMAppendBasicBlockInContext(ctx, fn, "merge");

LLVMBuildCondBr(bld, condition, then_bb, else_bb);

// Then block
LLVMPositionBuilderAtEnd(bld, then_bb);
// ... build then body ...
LLVMBuildBr(bld, merge_bb);

// Else block
LLVMPositionBuilderAtEnd(bld, else_bb);
// ... build else body ...
LLVMBuildBr(bld, merge_bb);

// Resume at merge
LLVMPositionBuilderAtEnd(bld, merge_bb);

Switch Statement

LLVMBasicBlockRef default_bb = LLVMAppendBasicBlockInContext(ctx, fn, "default");
LLVMValueRef sw = LLVMBuildSwitch(bld, val, default_bb, /*num_cases=*/2);

LLVMAddCase(sw, LLVMConstInt(i32, 1, 0), case_1_bb);
LLVMAddCase(sw, LLVMConstInt(i32, 2, 0), case_2_bb);

While Loop

LLVMBasicBlockRef cond_bb = LLVMAppendBasicBlockInContext(ctx, fn, "while.cond");
LLVMBasicBlockRef body_bb = LLVMAppendBasicBlockInContext(ctx, fn, "while.body");
LLVMBasicBlockRef end_bb  = LLVMAppendBasicBlockInContext(ctx, fn, "while.end");

LLVMBuildBr(bld, cond_bb);

// Condition
LLVMPositionBuilderAtEnd(bld, cond_bb);
LLVMValueRef cond = /* ... evaluate condition ... */;
LLVMBuildCondBr(bld, cond, body_bb, end_bb);

// Body
LLVMPositionBuilderAtEnd(bld, body_bb);
/* ... build body ... */
LLVMBuildBr(bld, cond_bb); // Loop back

// End
LLVMPositionBuilderAtEnd(bld, end_bb);

Do / While Loop

Similar to While, but bypasses the condition on the first entry.

LLVMBasicBlockRef body_bb = LLVMAppendBasicBlockInContext(ctx, fn, "do.body");
LLVMBasicBlockRef cond_bb = LLVMAppendBasicBlockInContext(ctx, fn, "do.cond");
LLVMBasicBlockRef end_bb  = LLVMAppendBasicBlockInContext(ctx, fn, "do.end");

LLVMBuildBr(bld, body_bb); // Enter body directly

// Body
LLVMPositionBuilderAtEnd(bld, body_bb);
/* ... build body ... */
LLVMBuildBr(bld, cond_bb);

// Condition
LLVMPositionBuilderAtEnd(bld, cond_bb);
LLVMValueRef cond = /* ... evaluate condition ... */;
LLVMBuildCondBr(bld, cond, body_bb, end_bb); 

LLVMPositionBuilderAtEnd(bld, end_bb);

For Loop

A for loop is effectively a block sequence: Init → Cond → Body → Step → Cond.

/* ... evaluate initializer ... */

LLVMBasicBlockRef cond_bb = LLVMAppendBasicBlockInContext(ctx, fn, "for.cond");
LLVMBasicBlockRef body_bb = LLVMAppendBasicBlockInContext(ctx, fn, "for.body");
LLVMBasicBlockRef inc_bb  = LLVMAppendBasicBlockInContext(ctx, fn, "for.inc");
LLVMBasicBlockRef end_bb  = LLVMAppendBasicBlockInContext(ctx, fn, "for.end");

LLVMBuildBr(bld, cond_bb); // Jump to cond

// Condition
LLVMPositionBuilderAtEnd(bld, cond_bb);
LLVMValueRef cond = /* ... */;
LLVMBuildCondBr(bld, cond, body_bb, end_bb);

// Body
LLVMPositionBuilderAtEnd(bld, body_bb);
/* ... */
LLVMBuildBr(bld, inc_bb);

// Increment
LLVMPositionBuilderAtEnd(bld, inc_bb);
/* ... e.g., i = i + 1 ... */
LLVMBuildBr(bld, cond_bb); // Loop back

LLVMPositionBuilderAtEnd(bld, end_bb);

10 Verification

Always verify before optimizing or emitting code — malformed IR otherwise fails silently or crashes deep in a pass.

char *err = NULL;
if (LLVMVerifyModule(mod, LLVMPrintMessageAction, &err)) {
    fprintf(stderr, "Invalid IR: %s\n", err);
}
LLVMDisposeMessage(err);

Dump IR for debugging

LLVMDumpModule(mod);                 // to stderr
char *ir = LLVMPrintModuleToString(mod);
LLVMPrintModuleToFile(mod, "out.ll", &err);

11 Optimization Passes (New Pass Manager)

#include <llvm-c/Transforms/PassBuilder.h>

LLVMPassBuilderOptionsRef opts = LLVMCreatePassBuilderOptions();
LLVMErrorRef e = LLVMRunPasses(mod, "default<O2>", targetMachine, opts);
if (e) {
    char *msg = LLVMGetErrorMessage(e);
    fprintf(stderr, "%s\n", msg);
    LLVMDisposeErrorMessage(msg);
}
LLVMDisposePassBuilderOptions(opts);
Common pipelines: default<O0>, default<O1>, default<O2>, default<O3>, default<Os>, or a custom comma-separated list like mem2reg,instcombine,simplifycfg.

12 Target Machine & Object Code Emission

LLVMInitializeAllTargetInfos();
LLVMInitializeAllTargets();
LLVMInitializeAllTargetMCs();
LLVMInitializeAllAsmParsers();
LLVMInitializeAllAsmPrinters();
// Or Native-only variants for a single-target compiler:
// LLVMInitializeNativeTarget(); LLVMInitializeNativeAsmPrinter();

char *triple = LLVMGetDefaultTargetTriple();
LLVMTargetRef target;
char *err = NULL;
LLVMGetTargetFromTriple(triple, &target, &err);

LLVMTargetMachineRef tm = LLVMCreateTargetMachine(
    target, triple, "generic", "",
    LLVMCodeGenLevelDefault, LLVMRelocPIC, LLVMCodeModelDefault);

LLVMSetModuleDataLayout(mod, LLVMCreateTargetDataLayout(tm));
LLVMSetTarget(mod, triple);

// Emit object file
LLVMTargetMachineEmitToFile(tm, mod, "out.o", LLVMObjectFile, &err);
// Or LLVMAssemblyFile for .s

LLVMDisposeTargetMachine(tm);
LLVMDisposeMessage(triple);
Then link with the system linker or simply clang out.o -o program.

13 JIT Execution (for interpreters / REPLs)

Legacy MCJIT — simple, single-module

#include <llvm-c/ExecutionEngine.h>

LLVMLinkInMCJIT();
LLVMInitializeNativeTarget();
LLVMInitializeNativeAsmPrinter();

LLVMExecutionEngineRef engine;
char *err = NULL;
LLVMCreateExecutionEngineForModule(&engine, mod, &err);

LLVMGenericValueRef args[] = { LLVMCreateGenericValueOfInt(i32, 5, 0) };
LLVMGenericValueRef result = LLVMRunFunction(engine, fn, 1, args);
int r = (int)LLVMGenericValueToInt(result, 0);

LLVMDisposeExecutionEngine(engine); // also disposes mod
Modern ORC v2 / LLJIT (llvm-c/LLJIT.h, llvm-c/Orc.h) is preferred for real JIT compilers — supports lazy compilation, multi-module setups, and symbol resolution across modules.

14 Error Handling

Newer APIs return LLVMErrorRef instead of the older char* + bool pattern.

LLVMErrorRef err = SomeCall(...);
if (err) {
    char *msg = LLVMGetErrorMessage(err); // consumes err
    fprintf(stderr, "%s\n", msg);
    LLVMDisposeErrorMessage(msg);
}
Older APIs use a char **errMsg out-parameter, freed with LLVMDisposeMessage.

16 Anatomy of a Full-Fledged Compiler

LLVM-C only covers IR generation, optimization, and codegen — the middle and back end. Everything else is on you.

source→ lexer→ tokens→ parser→ AST→ sema→ typed AST→ LLVM-C codegen→ LLVM IR→ verify→ optimize→ target machine→ object file→ linker→ executable
Front End · you write this

No LLVM involvement

  • Lexer — text → tokens; comments, escapes, literals, line/col tracking
  • Parser — tokens → AST; recursive descent or Pratt parsing
  • AST / IR structures — nodes for expressions, statements, declarations
  • Semantic analysis — symbol tables, scoping, type checking, name resolution, const-folding
  • Diagnostics engine — source-located errors/warnings, ideally with recovery
Middle End · LLVM-C does the work

Sections 03–11 apply here

  • AST → IR lowering — walk the AST, emit via LLVMBuilder*
  • IR-level correctness — LLVMVerifyModule
  • Optimization — new-PM pass pipelines
Back End

Sections 12–13 apply here

  • Target selection & codegen — TargetMachine
  • Object / assembly emission
  • Linking — system linker, lld, or shell out to clang/cc; link a runtime if needed (GC, strings, exceptions)
Supporting infrastructure

Ships the product

  • Runtime library — GC, closures, dynamic dispatch, primitives, unwinding
  • Standard library — language-level built-ins
  • Driver / CLI — orchestrates lex→parse→sema→codegen→link; flags like -O0..3, -o, -c, -emit-llvm
  • Debug info — DWARF emission for gdb/lldb
  • Testing infra — golden IR tests, FileCheck-style, end-to-end execution
  • JIT / incremental mode — optional, for REPLs or scripting languages

17 Memory Management Notes

  • Most LLVMDispose* functions release the corresponding object; disposing a Module or ExecutionEngine that owns it also frees owned Functions/BasicBlocks/Values.
  • Strings returned by LLVMPrintModuleToString, error messages, and target triples must be freed with LLVMDisposeMessage.
  • LLVMContextRef should outlive all Types/Values created within it.
  • Values (LLVMValueRef) are not individually disposed — they're owned by their parent Module/Function.

18 Quick Reference: Minimal Compilable Program

#include <llvm-c/Core.h>
#include <llvm-c/Analysis.h>
#include <llvm-c/TargetMachine.h>

int main() {
    LLVMContextRef ctx = LLVMContextCreate();
    LLVMModuleRef mod = LLVMModuleCreateWithNameInContext("m", ctx);
    LLVMBuilderRef bld = LLVMCreateBuilderInContext(ctx);

    LLVMTypeRef i32 = LLVMInt32TypeInContext(ctx);
    LLVMTypeRef params[] = { i32, i32 };
    LLVMTypeRef fnTy = LLVMFunctionType(i32, params, 2, 0);
    LLVMValueRef fn = LLVMAddFunction(mod, "add", fnTy);

    LLVMBasicBlockRef entry = LLVMAppendBasicBlockInContext(ctx, fn, "entry");
    LLVMPositionBuilderAtEnd(bld, entry);
    LLVMValueRef sum = LLVMBuildAdd(bld, LLVMGetParam(fn, 0), LLVMGetParam(fn, 1), "sum");
    LLVMBuildRet(bld, sum);

    char *err = NULL;
    LLVMVerifyModule(mod, LLVMPrintMessageAction, &err);
    LLVMDisposeMessage(err);

    LLVMDumpModule(mod); // prints IR to stderr

    LLVMDisposeBuilder(bld);
    LLVMDisposeModule(mod);
    LLVMContextDispose(ctx);
    return 0;
}
